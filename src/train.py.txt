import os
import pandas as pd
import joblib

from sklearn.model_selection import train_test_split
from sklearn.neighbors import KNeighborsClassifier
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import (
    accuracy_score,
    confusion_matrix,
    classification_report
)

from .features import FEATURE_COLUMNS


def get_model_paths():

    base_dir = os.path.dirname(
        os.path.dirname(
            os.path.abspath(__file__)
        )
    )

    return {
        "training_data": os.path.join(
            base_dir,
            "dataset",
            "training_dataset.csv"
        ),

        "models_dir": os.path.join(
            base_dir,
            "models"
        ),

        "knn": os.path.join(
            base_dir,
            "models",
            "knn_model.pkl"
        ),

        "random_forest": os.path.join(
            base_dir,
            "models",
            "random_forest_model.pkl"
        )
    }


def train_models():

    paths = get_model_paths()

    data_path = paths["training_data"]

    if not os.path.exists(data_path):

        print(
            f"Training dataset not found:\n"
            f"{data_path}"
        )

        return None, None

    df = pd.read_csv(data_path)

    if df.empty or "Label" not in df.columns:

        print("Invalid or empty training dataset.")

        return None, None

    # Make sure required features exist.
    missing = [
        col
        for col in FEATURE_COLUMNS
        if col not in df.columns
    ]

    if missing:

        print(
            "Training dataset is missing features:"
        )

        for col in missing:
            print(f" - {col}")

        return None, None

    X = df[FEATURE_COLUMNS]
    y = df["Label"].astype(int)

    print("==================================================")
    print("        MACHINE LEARNING MODEL TRAINING")
    print("==================================================")

    print(
        f"Total dataset size: {len(df)} samples"
    )

    print(
        f"Legitimate (0): {(y == 0).sum()}"
    )

    print(
        f"Evil Twin (1): {(y == 1).sum()}"
    )

    # ---------------------------------------------------------
    # Train/Test split
    # ---------------------------------------------------------

    X_train, X_test, y_train, y_test = train_test_split(
        X,
        y,
        test_size=0.20,
        random_state=42,
        stratify=y
    )

    # ---------------------------------------------------------
    # KNN
    # ---------------------------------------------------------

    knn = make_pipeline(
        StandardScaler(),
        KNeighborsClassifier(
            n_neighbors=5,
            weights="distance"
        )
    )

    knn.fit(
        X_train,
        y_train
    )

    knn_pred = knn.predict(X_test)

    knn_acc = (
        accuracy_score(
            y_test,
            knn_pred
        ) * 100
    )

    # ---------------------------------------------------------
    # Random Forest
    # ---------------------------------------------------------

    rf = RandomForestClassifier(
        n_estimators=100,
        max_depth=8,
        random_state=42,
        class_weight="balanced"
    )

    rf.fit(
        X_train,
        y_train
    )

    rf_pred = rf.predict(X_test)

    rf_acc = (
        accuracy_score(
            y_test,
            rf_pred
        ) * 100
    )

    # ---------------------------------------------------------
    # Results
    # ---------------------------------------------------------

    print("\n--------------------------------------------------")
    print("             MODEL EVALUATION")
    print("--------------------------------------------------")

    print(
        f"KNN Accuracy          : {knn_acc:.2f}%"
    )

    print(
        f"Random Forest Accuracy: {rf_acc:.2f}%"
    )

    print("\n=== KNN Confusion Matrix ===")

    print(
        confusion_matrix(
            y_test,
            knn_pred
        )
    )

    print("\n=== Random Forest Confusion Matrix ===")

    print(
        confusion_matrix(
            y_test,
            rf_pred
        )
    )

    print("\n=== Random Forest Classification Report ===")

    print(
        classification_report(
            y_test,
            rf_pred,
            target_names=[
                "Legitimate (0)",
                "Evil Twin (1)"
            ],
            zero_division=0
        )
    )

    # ---------------------------------------------------------
    # Save models
    # ---------------------------------------------------------

    os.makedirs(
        paths["models_dir"],
        exist_ok=True
    )

    joblib.dump(
        knn,
        paths["knn"]
    )

    joblib.dump(
        rf,
        paths["random_forest"]
    )

    print("--------------------------------------------------")

    print(
        f"KNN model saved:\n"
        f"{paths['knn']}"
    )

    print(
        f"Random Forest model saved:\n"
        f"{paths['random_forest']}"
    )

    print(
        "\nTraining completed."
    )

    print(
        "These models can now be reused during normal scans."
    )

    print("==================================================")

    return knn, rf


if __name__ == "__main__":
    train_models()
