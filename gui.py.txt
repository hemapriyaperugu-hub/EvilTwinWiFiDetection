def clear_table():
    """Remove all networks from the table."""

    for item in network_table.get_children():
        network_table.delete(item)


def update_output(text):
    """Display terminal output inside GUI."""

    output_box.config(state="normal")

    output_box.insert("end", text)

    output_box.see("end")

    output_box.config(state="disabled")


def load_networks_from_csv():
    """
    Read the Wi-Fi dataset created by main.py
    and display the detected networks.
    """

    clear_table()

    csv_path = os.path.join(
        PROJECT_FOLDER,
        "dataset",
        "wifi_dataset.csv"
    )

    if not os.path.exists(csv_path):
        return

    try:

        with open(
            csv_path,
            "r",
            encoding="utf-8-sig",
            newline=""
        ) as file:

            reader = csv.DictReader(file)

            rows = list(reader)

        for number, row in enumerate(rows, start=1):

            ssid = row.get("SSID", "--")
            bssid = row.get("BSSID", "--")
            signal = row.get("Signal", "--")
            channel = row.get("Channel", "--")
            security = row.get("Security", "--")

            # Convert empty values
            if not ssid:
                ssid = "<Hidden Network>"

            if not bssid:
                bssid = "--"

            if not signal:
                signal = "--"

            if not channel:
                channel = "--"

            if not security:
                security = "--"

            network_table.insert(
                "",
                "end",
                values=(
                    number,
                    ssid,
                    bssid,
                    signal,
                    channel,
                    security,
                    "ANALYZED"
                )
            )

        network_count.config(
            text=f"Networks found: {len(rows)}"
        )

    except Exception as error:

        update_output(
            f"\n[GUI] Could not read dataset: {error}\n"
        )


def show_final_result(output):
    """Determine the final result from main.py output."""

    text = output.upper()

    # Suspicious result has priority
    if (
        "SUSPICIOUS NETWORK" in text
        or "POSSIBLE EVIL TWIN" in text
    ):

        result_label.config(
            text="⚠  SUSPICIOUS NETWORK",
            fg=RED
        )

        result_description.config(
            text=(
                "Possible Evil Twin detected. "
                "Avoid connecting until the network is verified."
            )
        )

        return

    # Legitimate result
    if (
        "LEGITIMATE NETWORK" in text
        or "SAFE NETWORK" in text
    ):

        result_label.config(
            text="✓  SAFE NETWORK",
            fg=GREEN
        )

        result_description.config(
            text=(
                "No suspicious indicators were detected "
                "by the current detection models."
            )
        )

        return

    # Completed but result wasn't detected
    result_label.config(
        text="SCAN COMPLETED",
        fg=BLUE
    )

    result_description.config(
        text=(
            "The scan completed successfully. "
            "Check the analysis output below."
        )
    )


def scan_finished(output, return_code):
    """Runs after main.py finishes."""

    global scan_running

    scan_running = False

    scan_button.config(
        state="normal",
        text="  SCAN WI-FI NETWORKS  "
    )

    status_dot.config(
        fg=GREEN
    )

    status_label.config(
        text="Wi-Fi Adapter Ready"
    )

    if return_code != 0:

        scan_status.config(
            text="Scan Failed",
            fg=RED
        )

        result_label.config(
            text="SCAN ERROR",
            fg=RED
        )

        result_description.config(
            text="Check the output panel below for details."
        )

        return

    scan_status.config(
        text="Scan Completed",
        fg=GREEN
    )

    # Update network table
    load_networks_from_csv()

    # Show result
    show_final_result(output)


def run_main_program():
    """
    Execute the existing main.py.
    """

    try:

        # Use the same Python interpreter
        # that is running this GUI.
        python_executable = sys.executable

        process = subprocess.Popen(
            [
                python_executable,
                "main.py"
            ],

            cwd=PROJECT_FOLDER,

            stdout=subprocess.PIPE,
            stderr=subprocess.STDOUT,

            text=True,

            bufsize=1
        )

        complete_output = ""

        # Read main.py output line by line
        for line in process.stdout:

            complete_output += line

            root.after(
                0,
                update_output,
                line
            )

        process.wait()

        root.after(
            0,
            scan_finished,
            complete_output,
            process.returncode
        )

    except Exception as error:

        root.after(
            0,
            scan_error,
            str(error)
        )


def start_scan():
    """Start Wi-Fi scanning."""

    global scan_running

    if scan_running:
        return

    scan_running = True

    # Clear previous information
    clear_table()

    output_box.config(state="normal")
    output_box.delete("1.0", "end")
    output_box.config(state="disabled")

    # Update GUI
    scan_button.config(
        state="disabled",
        text="  SCANNING...  "
    )

    scan_status.config(
        text="Scanning nearby networks...",
        fg=YELLOW
    )

    status_dot.config(
        fg=YELLOW
    )

    status_label.config(
        text="Scanning Wi-Fi networks"
    )

    network_count.config(
        text="Networks found: --"
    )

    result_label.config(
        text="ANALYZING...",
        fg=YELLOW
    )

    result_description.config(
        text=(
            "Scanning with the built-in Wi-Fi adapter "
            "and running the detection models."
        )
    )

    # Run main.py in background
    thread = threading.Thread(
        target=run_main_program,
        daemon=True
    )

    thread.start()
